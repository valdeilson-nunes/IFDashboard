from flask import Blueprint, request, jsonify
from app import db
from app.models import Usuario, Publicacao, Notificacao, DEV_EMAIL

rotas_autenticacao = Blueprint('auth', __name__)


def obter_ator():
    user_id = request.headers.get('X-User-Id')
    if not user_id:
        return None
    try:
        return db.session.get(Usuario, int(user_id))
    except (ValueError, TypeError):
        return None


@rotas_autenticacao.route('/login', methods=['POST'])
def fazer_login():
    dados = request.get_json() or {}
    email = dados.get('email', '').strip().lower()
    senha = dados.get('password', '').strip()

    if not email or not senha:
        return jsonify({'error': 'E-mail e senha sÃ£o obrigatÃ³rios.'}), 400

    usuario = Usuario.query.filter_by(email=email).first()
    if not usuario:
        return jsonify({'error': 'UsuÃ¡rio nÃ£o encontrado. Realize seu cadastro primeiro.'}), 404

    if not usuario.verificar_senha(senha):
        return jsonify({'error': 'Senha incorreta.'}), 401

    return jsonify({'message': 'Login realizado com sucesso.', 'user': usuario.para_dicionario()}), 200

@rotas_autenticacao.route('/register', methods=['POST'])
def cadastrar_usuario():
    dados = request.get_json() or {}
    nome = dados.get('name', '').strip()
    email = dados.get('email', '').strip().lower()
    senha = dados.get('password', '').strip()

    if not nome:
        return jsonify({'error': 'Nome completo Ã© obrigatÃ³rio.'}), 400
    if not email:
        return jsonify({'error': 'E-mail institucional Ã© obrigatÃ³rio.'}), 400
    if email == DEV_EMAIL:
        return jsonify({'error': 'Este e-mail Ã© reservado pelo sistema.'}), 403
    if len(senha) < 6:
        return jsonify({'error': 'A senha deve conter no mÃ­nimo 6 caracteres.'}), 400

@rotas_autenticacao.route('/users', methods=['GET'])
def listar_usuarios():
    ator = obter_ator()
    if not ator or not ator.eh_admin_ou_dev:
        return jsonify({'error': 'Acesso negado.'}), 403

    usuarios = Usuario.query.filter(
        Usuario.email != DEV_EMAIL,
        Usuario.role != 'dev'
    ).order_by(Usuario.name.asc()).all()

    resultado = []
    for u in usuarios:
        d = u.para_dicionario()
        d['posts_count'] = Publicacao.query.filter(
            (Publicacao.user_id == u.id) | (Publicacao.author == u.name)
        ).count()
        resultado.append(d)

    return jsonify(resultado), 200


@rotas_autenticacao.route('/users/<int:target_id>', methods=['PUT'])
def atualizar_usuario(target_id):
    ator = obter_ator()
    if not ator or not ator.eh_admin_ou_dev:
        return jsonify({'error': 'Acesso negado.'}), 403

    alvo = db.get_or_404(Usuario, target_id)
    if alvo.eh_dev:
        return jsonify({'error': 'Este usuÃ¡rio nÃ£o pode ser modificado.'}), 403

    dados = request.get_json() or {}
    if dados.get('name', '').strip():
        alvo.name = dados['name'].strip()

    if dados.get('email', '').strip():
        novo_email = dados['email'].strip().lower()
        if novo_email == DEV_EMAIL:
            return jsonify({'error': 'E-mail reservado.'}), 403
        if Usuario.query.filter(Usuario.email == novo_email, Usuario.id != alvo.id).first():
            return jsonify({'error': 'JÃ¡ existe outro usuÃ¡rio com este e-mail.'}), 409
        alvo.email = novo_email

    if dados.get('role') in ('basico', 'admin'):
        alvo.role = dados['role']

    if dados.get('password', '').strip():
        nova_senha = dados['password'].strip()
        if len(nova_senha) < 6:
            return jsonify({'error': 'A nova senha deve ter no mÃ­nimo 6 caracteres.'}), 400
        alvo.definir_senha(nova_senha)

    db.session.commit()
    return jsonify({'message': 'Dados atualizados com sucesso.', 'user': alvo.para_dicionario()}), 200

import os
import uuid
from datetime import datetime, timedelta
from flask import Blueprint, request, jsonify, current_app
from werkzeug.utils import secure_filename
from app import db
from app.models import Publicacao, Evento, Usuario, Notificacao
from app.routes.auth import obter_ator

EXTENSOES_PERMITIDAS = {'png', 'jpg', 'jpeg', 'webp', 'gif'}
rotas_publicacoes = Blueprint('posts', __name__)


def eh_dono_post(usuario, publicacao):
    if not usuario:
        return False
    if publicacao.user_id and publicacao.user_id == usuario.id:
        return True
    return (publicacao.author or '').strip().lower() == (usuario.name or '').strip().lower()


def arquivo_permitido(nome_arquivo):
    return '.' in nome_arquivo and nome_arquivo.rsplit('.', 1)[1].lower() in EXTENSOES_PERMITIDAS


def notificar_autor_moderacao(ator, publicacao, titulo, mensagem, icone):
    if not (ator.eh_admin_ou_dev and not eh_dono_post(ator, publicacao)):
        return
    destinatario_id = publicacao.user_id
    if not destinatario_id and publicacao.author:
        autor_obj = Usuario.query.filter_by(name=publicacao.author).first()
        destinatario_id = autor_obj.id if autor_obj else None
    if destinatario_id:
        db.session.add(Notificacao(
            user_id=destinatario_id,
            titulo=titulo,
            mensagem=mensagem,
            icone=icone
        ))


@rotas_publicacoes.route('', methods=['GET'])
def listar_publicacoes():
    categoria = request.args.get('category')
    termo_busca = request.args.get('search')
    ordenacao = request.args.get('sort', 'recent')

    consulta = Publicacao.query
    if categoria and categoria != 'todos':
        consulta = consulta.filter_by(category=categoria)

    if termo_busca:
        filtro = f"%{termo_busca}%"
        consulta = consulta.filter(
            Publicacao.title.ilike(filtro) |
            Publicacao.description.ilike(filtro) |
            Publicacao.location.ilike(filtro)
        )

    if ordenacao == 'likes':
        consulta = consulta.order_by(Publicacao.is_pinned.desc(), Publicacao.likes_count.desc(), Publicacao.created_at.desc())
    else:
        consulta = consulta.order_by(Publicacao.is_pinned.desc(), Publicacao.created_at.desc())

    return jsonify([p.para_dicionario() for p in consulta.all()]), 200

@rotas_publicacoes.route('', methods=['POST'])
def criar_publicacao():
    ator = obter_ator()
    if not ator or ator.role == 'guest':
        return jsonify({'error': 'Visitantes nÃ£o tÃªm permissÃ£o para publicar.'}), 403

    if request.is_json:
        dados = request.get_json() or {}
        url_imagem = dados.get('image_url')
    else:
        dados = request.form.to_dict()
        url_imagem = dados.get('image_url')
        arquivo = request.files.get('image')
        if arquivo and arquivo.filename:
            if not arquivo_permitido(arquivo.filename):
                return jsonify({'error': 'Formato invÃ¡lido. Use PNG, JPG, JPEG, WEBP ou GIF.'}), 400
            nome_unico = f"{uuid.uuid4().hex}_{secure_filename(arquivo.filename)}"
            arquivo.save(os.path.join(current_app.config['UPLOAD_FOLDER'], nome_unico))
            url_imagem = f"{request.host_url.rstrip('/')}/uploads/{nome_unico}"

    categoria = dados.get('category')
    if not dados.get('title') or not categoria:
        return jsonify({'error': 'TÃ­tulo e categoria sÃ£o obrigatÃ³rios.'}), 400

    if not ator.eh_admin_ou_dev and categoria not in ('achados', 'servicos'):
        return jsonify({'error': 'UsuÃ¡rios bÃ¡sicos sÃ³ podem publicar em Achados e Perdidos ou ServiÃ§os e Vendas.'}), 403

    prazo_formatado = dados.get('deadline')
    dt_inicio = dt_fim = None

    if categoria in ('pesquisa', 'editais', 'avisos'):
        inicio_str = (dados.get('start_date') or '').strip()
        fim_str = (dados.get('end_date') or '').strip()
        if not inicio_str:
            return jsonify({'error': 'A data de inÃ­cio Ã© obrigatÃ³ria para esta categoria.'}), 400
        try:
            dt_inicio = datetime.strptime(inicio_str, '%Y-%m-%d').date()
            dt_fim = datetime.strptime(fim_str, '%Y-%m-%d').date() if fim_str else dt_inicio
        except ValueError:
            return jsonify({'error': 'Formato de data invÃ¡lido. Use AAAA-MM-DD.'}), 400

        if dt_fim < dt_inicio:
            return jsonify({'error': 'A data de tÃ©rmino nÃ£o pode ser anterior Ã  de inÃ­cio.'}), 400

        prazo_formatado = (
            dt_inicio.strftime('%d/%m/%Y')
            if dt_inicio == dt_fim
            else f"{dt_inicio.strftime('%d/%m/%Y')} a {dt_fim.strftime('%d/%m/%Y')}"
        )

    nova = Publicacao(
        user_id=ator.id,
        category=categoria,
        sub_type=dados.get('sub_type'),
        title=dados.get('title').strip(),
        description=dados.get('description', '').strip(),
        location=dados.get('location', '').strip() or 'Campus Lapa',
        author=ator.name,
        price=dados.get('price'),
        deadline=prazo_formatado,
        contact_wpp=dados.get('contact_wpp'),
        image_url=url_imagem,
        is_pinned=bool(dados.get('is_pinned', False)) if ator.eh_admin_ou_dev else False
    )
    db.session.add(nova)
    db.session.flush()

    if dt_inicio and dt_fim:
        total_dias = (dt_fim - dt_inicio).days
        for i in range(total_dias + 1):
            dia = dt_inicio + timedelta(days=i)
            titulo_ev = nova.title
            if total_dias > 0:
                if i == 0:
                    titulo_ev = f"[InÃ­cio] {nova.title}"
                elif i == total_dias:
                    titulo_ev = f"[Encerramento] {nova.title}"

            db.session.add(Evento(
                category=nova.category,
                title=titulo_ev,
                description=nova.description,
                event_date=dia,
                post_id=nova.id
            ))

    db.session.commit()
    return jsonify(nova.para_dicionario()), 201

from datetime import datetime
from werkzeug.security import generate_password_hash, check_password_hash
from app import db

DEV_EMAIL = 'administrador@ifdashboard.com'


class Usuario(db.Model):
    __tablename__ = 'users'

    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)
    email = db.Column(db.String(120), unique=True, nullable=False)
    password_hash = db.Column(db.String(255), nullable=True)
    avatar = db.Column(db.String(255), nullable=True)
    role = db.Column(db.String(50), default='basico')  # guest, basico, admin, dev
    created_at = db.Column(db.DateTime, default=datetime.utcnow)

    def definir_senha(self, senha):
        self.password_hash = generate_password_hash(senha)

    def verificar_senha(self, senha):
        return bool(self.password_hash) and check_password_hash(self.password_hash, senha)

    @property
    def eh_admin_ou_dev(self):
        return self.role in ('admin', 'dev')

    @property
    def eh_dev(self):
        return self.role == 'dev' or self.email == DEV_EMAIL

    def para_dicionario(self):
        return {
            'id': self.id,
            'name': self.name,
            'email': self.email,
            'avatar': self.avatar,
            'role': 'basico' if self.role == 'student' else self.role,
            'created_at': self.created_at.isoformat() if self.created_at else None,
        }

def limpar_banco():
    with aplicacao.app_context():
        pasta_uploads = aplicacao.config['UPLOAD_FOLDER']
        if os.path.exists(pasta_uploads):
            for pub in Publicacao.query.filter(Publicacao.image_url.isnot(None)).all():
                if '/uploads/' in pub.image_url:
                    nome = pub.image_url.rsplit('/uploads/', 1)[-1]
                    caminho = os.path.join(pasta_uploads, nome)
                    if os.path.exists(caminho):
                        os.remove(caminho)

const URL_BASE_API = '/api';

function obterHeadersAuth(extra = {}) {
  const headers = { ...extra };
  try {
    const u = JSON.parse(localStorage.getItem('ifdashboard_user') || 'null');
    if (u && u.id) headers['X-User-Id'] = String(u.id);
  } catch (e) {}
  return headers;
}

async function requisicaoJson(url, opcoes = {}, erroPadrao = 'Erro na requisiÃ§Ã£o.') {
  const resposta = await fetch(url, opcoes);
  const dados = await resposta.json().catch(() => ({}));
  if (!resposta.ok) throw new Error(dados.error || erroPadrao);
  return dados;
}

const servicoApi = {
  obterPublicacoes(categoria = '', busca = '', ordenacao = 'recent') {
    const params = new URLSearchParams();
    if (categoria && categoria !== 'todos') params.append('category', categoria);
    if (busca) params.append('search', busca);
    if (ordenacao) params.append('sort', ordenacao);
    return requisicaoJson(`${URL_BASE_API}/posts?${params.toString()}`, {}, 'Erro ao buscar publicaÃ§Ãµes.');
  },

  obterPublicacaoPorId(id) {
    return requisicaoJson(`${URL_BASE_API}/posts/${id}`, {}, 'PublicaÃ§Ã£o nÃ£o encontrada.');
  },

  criarPublicacao(dadosPublicacao) {
    const ehFormData = dadosPublicacao instanceof FormData;
    return requisicaoJson(`${URL_BASE_API}/posts`, {
      method: 'POST',
      headers: ehFormData ? obterHeadersAuth() : obterHeadersAuth({ 'Content-Type': 'application/json' }),
      body: ehFormData ? dadosPublicacao : JSON.stringify(dadosPublicacao)
    }, 'Erro ao criar publicaÃ§Ã£o.');
  },

  editarPublicacao(id, dados) {
    return requisicaoJson(`${URL_BASE_API}/posts/${id}`, {
      method: 'PUT',
      headers: obterHeadersAuth({ 'Content-Type': 'application/json' }),
      body: JSON.stringify(dados)
    }, 'Erro ao editar publicaÃ§Ã£o.');
  },

  removerPublicacao(id) {
    return requisicaoJson(`${URL_BASE_API}/posts/${id}`, {
      method: 'DELETE',
      headers: obterHeadersAuth()
    }, 'Erro ao remover publicaÃ§Ã£o.');
  },

  alternarCurtida(id, acao = 'increment') {
    return requisicaoJson(`${URL_BASE_API}/posts/${id}/like`, {
      method: 'POST',
      headers: obterHeadersAuth({ 'Content-Type': 'application/json' }),
      body: JSON.stringify({ action: acao })
    }, 'Erro ao registrar curtida.');
  },

  obterEventos() {
    return requisicaoJson(`${URL_BASE_API}/events`, {}, 'Erro ao buscar eventos.');
  },

  criarEvento(dadosEvento) {
    return requisicaoJson(`${URL_BASE_API}/events`, {
      method: 'POST',
      headers: obterHeadersAuth({ 'Content-Type': 'application/json' }),
      body: JSON.stringify(dadosEvento)
    }, 'Erro ao criar evento.');
  },

  fazerLogin(email, senha) {
    return requisicaoJson(`${URL_BASE_API}/auth/login`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password: senha })
    }, 'Falha ao autenticar.');
  },

  cadastrarUsuario(nome, email, senha) {
    return requisicaoJson(`${URL_BASE_API}/auth/register`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name: nome, email, password: senha })
    }, 'Falha ao cadastrar usuÃ¡rio.');
  },

  entrarComoVisitante() {
    return requisicaoJson(`${URL_BASE_API}/auth/guest`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' }
    }, 'Falha ao entrar como visitante.');
  },

  listarUsuarios() {
    return requisicaoJson(`${URL_BASE_API}/auth/users`, {
      headers: obterHeadersAuth()
    }, 'Erro ao listar usuÃ¡rios.');
  },

  atualizarUsuarioAdmin(id, dados) {
    return requisicaoJson(`${URL_BASE_API}/auth/users/${id}`, {
      method: 'PUT',
      headers: obterHeadersAuth({ 'Content-Type': 'application/json' }),
      body: JSON.stringify(dados)
    }, 'Erro ao atualizar usuÃ¡rio.');
  },

  obterPostsUsuarioAdmin(id) {
    return requisicaoJson(`${URL_BASE_API}/auth/users/${id}/posts`, {
      headers: obterHeadersAuth()
    }, 'Erro ao carregar posts do usuÃ¡rio.');
  },

  async obterNotificacoes(userId) {
    try {
      return await requisicaoJson(`${URL_BASE_API}/auth/notifications/${userId}`, {
        headers: obterHeadersAuth()
      });
    } catch {
      return [];
    }
  },

  marcarNotificacoesLidas(userId) {
    return fetch(`${URL_BASE_API}/auth/notifications/${userId}/read`, {
      method: 'POST',
      headers: obterHeadersAuth()
    });
  },

  limparNotificacoesApi(userId) {
    return fetch(`${URL_BASE_API}/auth/notifications/${userId}`, {
      method: 'DELETE',
      headers: obterHeadersAuth()
    });
  }
};

    const atual = document.documentElement.getAttribute('data-bs-theme') || 'light';

  });
}

  if (!texto) return '';
  return String(texto).replace(/[&<>'"]/g, tag => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;'
  }[tag] || tag));

function formatarChaveData(data) {
  const ano = data.getFullYear();
  const mes = String(data.getMonth() + 1).padStart(2, '0');
  const dia = String(data.getDate()).padStart(2, '0');
  return `${ano}-${mes}-${dia}`;
}

async function iniciarCalendario() {
  await carregarEventosCalendario();

async function carregarEventosCalendario() {
  try {
    const eventos = await servicoApi.obterEventos();
    todosEventosCalendario = Array.isArray(eventos) ? eventos : [];
  } catch {
    todosEventosCalendario = [];
  }

  eventosPorData = new Map();
  todosEventosCalendario.forEach(ev => {
    if (!ev.event_date) return;
    const chave = ev.event_date.substring(0, 10);
    if (!eventosPorData.has(chave)) eventosPorData.set(chave, []);
    eventosPorData.get(chave).push(ev);
  });

  const prefixoAtual = `${dataCalendario.getFullYear()}-${String(dataCalendario.getMonth() + 1).padStart(2, '0')}`;
  const temNoMesAtual = Array.from(eventosPorData.keys()).some(k => k.startsWith(prefixoAtual));

  if (eventosPorData.size > 0 && !temNoMesAtual) {
    const chaveAlvo = Array.from(eventosPorData.keys()).sort()[0];
    const [ano, mes] = chaveAlvo.split('-');

function podeEditarPublicacao(post) {
  const u = obterUsuarioSessao();
  if (!u || u.role === 'guest') return false;
  if (u.role === 'admin' || u.role === 'dev') return true;
  const ehDono = (post.user_id && Number(post.user_id) === Number(u.id)) ||
                 ((post.author || '').trim().toLowerCase() === (u.name || '').trim().toLowerCase());
  if (!ehDono || !post.created_at) return false;
  const diffMin = (Date.now() - new Date(post.created_at + (post.created_at.endsWith('Z') ? '' : 'Z')).getTime()) / 60000;
  return diffMin <= 30;
}

function podeExcluirPublicacao(post) {
  const u = obterUsuarioSessao();
  if (!u || u.role === 'guest') return false;
  if (u.role === 'admin' || u.role === 'dev') return true;
  return (post.user_id && Number(post.user_id) === Number(u.id)) ||
         ((post.author || '').trim().toLowerCase() === (u.name || '').trim().toLowerCase());
}

async function aoCurtir(idPublicacao, botao) {
  if (ehConvidado()) {
    abrirModalCadastroConvidado();
    return;
  }

  const foiCurtido = publicacoesCurtidas.has(idPublicacao);
  try {
    const resultado = await servicoApi.alternarCurtida(idPublicacao, foiCurtido ? 'decrement' : 'increment');
    foiCurtido ? publicacoesCurtidas.delete(idPublicacao) : publicacoesCurtidas.add(idPublicacao);
    localStorage.setItem(obterChaveUsuario('ifdashboard_liked_posts'), JSON.stringify(Array.from(publicacoesCurtidas)));

    const postObj = listaPublicacoes.find(p => p.id === idPublicacao);
    if (postObj) postObj.likes_count = resultado.likes_count;

    if (botao) {
      const span = botao.querySelector('.contador-curtidas');
      if (span) span.textContent = resultado.likes_count;
      botao.classList.toggle('btn-danger', !foiCurtido);
      botao.classList.toggle('btn-outline-danger', foiCurtido);
      const icone = botao.querySelector('i');
      if (icone) icone.className = !foiCurtido ? 'bi bi-heart-fill' : 'bi bi-heart';
    }
    if (abaAtual === 'perfil') renderizarPerfil(abaPerfilAtual);
  } catch (erro) {
    exibirAlerta(erro.message, 'warning');
  }
}

async function curtirNoModal(idPublicacao) {
  await aoCurtir(idPublicacao, null);
  const publicacao = listaPublicacoes.find(p => p.id === idPublicacao);
  if (!publicacao) return;

  const estaCurtido = publicacoesCurtidas.has(idPublicacao);
  document.getElementById('postModalLikeCount').textContent = publicacao.likes_count || 0;
  document.getElementById('postModalLikeBtn').className = `btn ${estaCurtido ? 'btn-danger' : 'btn-outline-danger'} d-flex align-items-center gap-2`;
  document.getElementById('postModalLikeIcon').className = `bi ${estaCurtido ? 'bi-heart-fill' : 'bi-heart'}`;
  document.getElementById('postModalLikeText').textContent = estaCurtido ? 'Curtido' : 'Curtir';
  if (abaAtual !== 'perfil') renderizarPublicacoes(listaPublicacoes);
}

function alternarSalvar(idPublicacao, botao) {
  if (ehConvidado()) {
    abrirModalCadastroConvidado();
    return;
  }

  const foiSalvo = publicacoesSalvas.has(idPublicacao);
  if (foiSalvo) {
    publicacoesSalvas.delete(idPublicacao);
  } else {
    publicacoesSalvas.add(idPublicacao);
    adicionarNotificacao('Item Salvo', 'Uma publicaÃ§Ã£o foi adicionada aos seus itens salvos.', 'bi-bookmark-check-fill text-warning');
  }
  localStorage.setItem(obterChaveUsuario('ifdashboard_saved_posts'), JSON.stringify(Array.from(publicacoesSalvas)));

  const estaSalvo = !foiSalvo;
  if (botao) {
    botao.classList.toggle('btn-warning', estaSalvo);
    botao.classList.toggle('text-dark', estaSalvo);
    botao.classList.toggle('btn-outline-secondary', !estaSalvo);
    const icone = botao.querySelector('i');
    if (icone) icone.className = estaSalvo ? 'bi bi-bookmark-fill' : 'bi bi-bookmark';
  }
  if (idPublicacaoAtualDetalhes === idPublicacao) atualizarBotaoSalvarModal(estaSalvo);
  if (abaAtual === 'perfil') renderizarPerfil(abaPerfilAtual);
}

function aoSalvarNoModal() {

}

function previsualizarImagem(evento) {
  const arquivo = evento.target.files?.[0];
  if (!arquivo) return limparUploadImagem();

  if (!arquivo.type.startsWith('image/') || arquivo.size > 16 * 1024 * 1024) {
    exibirAlerta('Selecione uma imagem vÃ¡lida (PNG, JPG, WEBP ou GIF) de atÃ© 16MB.', 'warning');
    return limparUploadImagem();
  }

  const leitor = new FileReader();
  leitor.onload = e => {
    document.getElementById('postImagePreview').src = e.target.result;
    document.getElementById('postImagePreviewContainer').classList.remove('d-none');
  };
  leitor.readAsDataURL(arquivo);
}

function limparUploadImagem() {

  if (campo) campo.value = '';
  document.getElementById('postImagePreviewContainer')?.classList.add('d-none');

  if (img) img.src = '';
}

async function salvarUsuarioAdmin(evento) {

function adicionarNotificacao(titulo, mensagem, icone = 'bi-bell-fill text-primary') {
  notificacoesUsuario.unshift({
    id: Date.now(),

    mensagem,
    icone,
    data: 'Agora',
    lida: false
  });

}

async function marcarNotificacoesComoLidas() {
  notificacoesUsuario.forEach(n => { n.lida = true; });
  localStorage.setItem(obterChaveUsuario('ifdashboard_notifications'), JSON.stringify(notificacoesUsuario));
  const u = obterUsuarioSessao();
  if (u && u.id && u.role !== 'guest') await servicoApi.marcarNotificacoesLidas(u.id);
  renderizarPerfil('notificacoes');
}

async function limparNotificacoes() {
  notificacoesUsuario = [];
  localStorage.setItem(obterChaveUsuario('ifdashboard_notifications'), JSON.stringify(notificacoesUsuario));
  const u = obterUsuarioSessao();
  if (u && u.id && u.role !== 'guest') await servicoApi.limparNotificacoesApi(u.id);
  renderizarPerfil('notificacoes');
}
